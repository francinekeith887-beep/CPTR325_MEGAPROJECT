Requirement 3: User Accounts & Organizational Units

Step-by-Step Procedure:

  1. Create 8 OUs: COMTECH1A, COMTECH1B, COMTECH2A, COMTECH2B, COMTECH3A, COMTECH3B, COMTECH4A, COMTECH4B
  2. Automated OU Creation: PowerShell script executed successfully
  3. Create Teacher Accounts: 1 teacher per class (8 total)
  4. Create Student Accounts: 40 students per class (320 total)
  5. Verify Structure: ADUC → Check OU population & inheritance
# CREATE ORGANIZATIONAL UNITS
$ous = @("COMTECH1A","COMTECH1B","COMTECH2A","COMTECH2B","COMTECH3A","COMTECH3B","COMTECH4A","COMTECH4B")
foreach ($ou in $ous) {
    New-ADOrganizationalUnit -Name $ou -Path "DC=group2,DC=local"
}

# CREATE USERS (8 Teachers + 320 Students)
$classes = @("COMTECH1A","COMTECH1B","COMTECH2A","COMTECH2B","COMTECH3A","COMTECH3B","COMTECH4A","COMTECH4B")
foreach ($class in $classes) {
    # Teacher Account
    New-ADUser -Name "teacher_$class" `
    -SamAccountName "teacher_$class" `
    -UserPrincipalName "teacher_$class@group2.local" `
    -Path "OU=$class,DC=group2,DC=local" `
    -AccountPassword (ConvertTo-SecureString "P@ssw0rd123" -AsPlainText -Force) `
    -Enabled $true

    # 40 Student Accounts per Class
    for ($i=1; $i -le 40; $i++) {
        $num = "{0:D2}" -f $i
        New-ADUser -Name "stud${num}_$class" `
        -SamAccountName "stud${num}_$class" `
        -UserPrincipalName "stud${num}_$class@group2.local" `
        -Path "OU=$class,DC=group2,DC=local" `
        -AccountPassword (ConvertTo-SecureString "P@ssw0rd123" -AsPlainText -Force) `
        -Enabled $true
    }
}
