Requirement 5: Logon Time Restrictions per User

Verification (Exact Steps):

  1. Run PowerShell as Admin
  2. Open ADUC → CT1A OU
  3. Check Logon Hours
  4. Test login restrictions
  5. Enable auto logoff policy

🚀 Complete PowerShell Script:

Import-Module ActiveDirectory

function New-LogonArray {
    return New-Object byte[] 21
}

function Set-Hour {
    param($array, $day, $start, $end)
    for ($h = $start; $h -lt $end; $h++) {
        $index = ($day * 24) + $h
        $byte = [math]::Floor($index / 8)
        $bit = $index % 8
        $array[$byte] = $array[$byte] -bor (1 -shl $bit)
    }
}

# CT1A: Mon-Fri 00:00-09:00
$ct1a = New-LogonArray
for ($d=1; $d -le 5; $d++) { Set-Hour $ct1a $d 0 9 }
Get-ADUser -Filter * -SearchBase "OU=CT1A,DC=group1,DC=local" |
    ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct1a} }

# CT1B: Tue/Thu 00:00-09:00
$ct1b = New-LogonArray
Set-Hour $ct1b 2 0 9; Set-Hour $ct1b 4 0 9
Get-ADUser -Filter * -SearchBase "OU=CT1B,DC=group1,DC=local" |
    ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct1b} }

# CT2A: Mon-Wed 23:00-08:00
$ct2a = New-LogonArray
